function construirIndiceUnico() { indiceGeral.clear(); // 1. Indexa Personagens baseDados.personagens.forEach(p => { let chaveBusca = normalizar(p.nome); // Ex: "suzane" ou "dicionario" if (chaveBusca) { indiceGeral.set(chaveBusca, { tipo: "PERSONAGEM", palavra: p.nome, // Mantém o acento original para o Popup sub: `Papel: ${p.papel}`, significado: p.bio }); } }); // 2. Indexa Dicionário (Suportando acentos e sinônimos) baseDados.dicionario.forEach(d => { let chaveBusca = normalizar(d.termo); // Ex: "dicionario" acha "Dicionário" if (chaveBusca) { indiceGeral.set(chaveBusca, { tipo: "DICIONÁRIO", palavra: d.termo, // Exibe "Dicionário" com acento no popup sub: `Sinônimo / Categoria: ${d.categoria || 'Geral'}`, significado: d.definicao || d.sinonimo || "Termo registrado." }); } }); }
